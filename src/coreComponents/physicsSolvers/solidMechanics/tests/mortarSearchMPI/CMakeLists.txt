cmake_minimum_required(VERSION 3.16)
project(MPMMortarSearchStateTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MPM_TEST_MPI "Build actual MPI request/state fixture" OFF)
add_executable(testIndexedSearch testIndexedSearch.cpp)
add_executable(testStateValidation testStateValidation.cpp)
enable_testing()
add_test(NAME exact_indexed_search COMMAND testIndexedSearch)
add_test(NAME boundary_closure_state_audit COMMAND testStateValidation)
if(MPM_TEST_MPI)
  find_package(MPI REQUIRED COMPONENTS CXX)
  add_executable(testMPIState testMPIState.cpp)
  target_link_libraries(testMPIState PRIVATE MPI::MPI_CXX)
  foreach(N 1 2 4 8)
    add_test(NAME mpi_state_${N}
      COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${N}
              ${MPIEXEC_PREFLAGS} $<TARGET_FILE:testMPIState> ${MPIEXEC_POSTFLAGS})
  endforeach()
endif()
