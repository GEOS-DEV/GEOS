cmake_minimum_required(VERSION 3.16)
project(MPMMortarPerformanceTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MPM_TEST_LAPACK "Test the optional LP64 LAPACK backend" OFF)
option(MPM_TEST_MPI "Build the real MPI scheduler test" OFF)
find_package(Threads REQUIRED)
set(KERNELS "${CMAKE_CURRENT_LIST_DIR}/../../kernels")
add_library(mortarQR STATIC "${KERNELS}/MPMBoundaryMortarQR.cpp")
if(MPM_TEST_LAPACK)
  set(BLA_SIZEOF_INTEGER 4)
  find_package(BLAS REQUIRED)
  find_package(LAPACK REQUIRED)
  target_compile_definitions(mortarQR PRIVATE GEOS_MPM_MORTAR_LAPACK=1)
  target_link_libraries(mortarQR PUBLIC ${LAPACK_LIBRARIES} ${BLAS_LIBRARIES})
endif()
add_executable(testPerformance testPerformance.cpp)
target_link_libraries(testPerformance PRIVATE mortarQR Threads::Threads)
add_executable(testExisting ../testMPMBoundaryMortar.cpp)
enable_testing()
add_test(NAME existing_reference COMMAND testExisting)
add_test(NAME native_search_and_scheduler COMMAND testPerformance native)
if(MPM_TEST_LAPACK)
  add_test(NAME lapack_search_and_scheduler COMMAND testPerformance lapack)
endif()
if(MPM_TEST_MPI)
  find_package(MPI REQUIRED COMPONENTS CXX)
  add_executable(testMPI testMPI.cpp)
  target_link_libraries(testMPI PRIVATE mortarQR MPI::MPI_CXX)
endif()
